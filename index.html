from PySide6.QtWidgets import QApplication
from core import adb, scrcpy

class NayerControl(MainWindow):
    def connect(self, mode="usb"):
        device = adb.find_device(mode)
        if device:
            scrcpy.mirror(device, fps=60)
            self.status("conectado")
